    sheet.getRange("1:1").setFontWeight("bold").setBackground("#d9ead3");

    if (lastCol < 11) pSheet.getRange(1, 11).setValue("Stock_Qty");

function getAppData() {
  const products = getFirebaseData("appData/products") || [];
  const machines = getFirebaseData("appData/machines") || [];
  const mappings = getFirebaseData("mappings") || [];
  const settings = getFirebaseData("appData/settings") || { isShowPriceBForGuest: false, isShowPriceCForGuest: false };
  const manuals = getFirebaseData("appData/manuals") || [];
  const lots = getLotsFromJson();
  const purchaseOrders = getFirebaseData("appData/purchaseOrders") || [];
  
  return {
    products: products,
    machines: machines,
    mappings: mappings,
    settings: settings,
    manuals: manuals,
    lots: lots,
    purchaseOrders: purchaseOrders
  };
}

// ฟังก์ชันอ่านข้อมูลจาก Google Sheets และ JSON แยกต่างๆ แล้วนำมารวมกันเป็นก้อนเดียว เซฟลง Drive และคืนค่า
function rebuildAndSaveAppDataJson(throwError = false) {
  initDatabase();
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  
  // ดึงข้อมูล Products
  const prodSheet = ss.getSheetByName("Products");
  const prodData = prodSheet.getDataRange().getValues();
  const products = [];
  if (prodData.length > 1) {
    for (let i = 1; i < prodData.length; i++) {
      let r = prodData[i];
      products.push({
        id: r[0], name: r[1], unit: r[2], cost: r[3],
        price_a: r[4], price_b: r[5], price_c: r[6],
        category: r[7], note: r[8], image_url: r[9],
        stock_qty: (r[10] !== undefined && r[10] !== "") ? parseFloat(r[10]) : 0,
        group: r[11] || "",
        supplier: r[12] || "",
        storage: r[13] || ""
      });
    }
  }

  // ดึงข้อมูล Machines
  const machSheet = ss.getSheetByName("Machines");
  const machData = machSheet.getDataRange().getValues();
  const machines = [];
  if (machData.length > 1) {
    for (let i = 1; i < machData.length; i++) {
      machines.push({ 
        id: machData[i][0], 
        name: machData[i][1], 
        image_url: machData[i][2] || '',
        cost: machData[i][3] || 0,
        price_a: machData[i][4] || 0,
        price_b: machData[i][5] || 0,
        price_c: machData[i][6] || 0,
        note: machData[i][7] || '',
        group: machData[i][8] || '',
        supplier: machData[i][9] || '',
        storage: machData[i][10] || ''
      });
    }
  }

  // ดึงข้อมูล Mapping จาก JSON ใน Google Drive
  let mappings = getMappingsFromJson();
  const machineIds = new Set(machines.map(m => String(m.id).trim()));
  const productIds = new Set(products.map(p => String(p.id).trim()));
  const initialMappingsLength = mappings.length;

  mappings = mappings.filter(m => {
    return machineIds.has(String(m.machine_id).trim()) && 
           productIds.has(String(m.product_id).trim());
  });
  if (mappings.length < initialMappingsLength) {
    try {
      saveMappingsToJson(mappings);
    } catch (e) {
      Logger.log("ไม่สามารถเซฟข้อมูลการจับคู่ที่กรองใหม่ได้: " + e.toString());
    }
  }

  // ดึงข้อมูล Settings
  const settingsSheet = ss.getSheetByName("Settings");
  const settings = { isShowPriceBForGuest: false, isShowPriceCForGuest: false };
  if (settingsSheet) {
    const settingsData = settingsSheet.getDataRange().getValues();
    if (settingsData.length > 1) {
      for (let i = 1; i < settingsData.length; i++) {
        let key = settingsData[i][0];
        let val = settingsData[i][1];
        if (key === "isShowPriceBForGuest") settings.isShowPriceBForGuest = (String(val).toLowerCase() === "true");
        if (key === "isShowPriceCForGuest") settings.isShowPriceCForGuest = (String(val).toLowerCase() === "true");
      }
    }
  }

  // ดึงข้อมูล Manuals
  const manSheet = ss.getSheetByName("Manuals");
  const manuals = [];
  if (manSheet) {
    const manData = manSheet.getDataRange().getValues();
    if (manData.length > 1) {
      for (let i = 1; i < manData.length; i++) {
        manuals.push({
          id: manData[i][0],
          title: manData[i][1],
          description: manData[i][2] || '',
          file_url: manData[i][3] || '',
          file_type: manData[i][4] || '',
          uploaded_at: manData[i][5] || ''
        });
      }
    }
  }

  // ดึงข้อมูล PurchaseOrders
  const poSheet = ss.getSheetByName("PurchaseOrders");
  const purchaseOrders = [];
  if (poSheet) {
    const poData = poSheet.getDataRange().getValues();
    if (poData.length > 1) {
      const timeZone = ss.getSpreadsheetTimeZone();
      for (let i = 1; i < poData.length; i++) {
        let orderDateStr = "";
        if (poData[i][4]) {
          orderDateStr = (poData[i][4] instanceof Date) ? Utilities.formatDate(poData[i][4], timeZone, "yyyy-MM-dd") : String(poData[i][4]);
        }
        let lastReceivedDateStr = "";
        if (poData[i][7]) {
          lastReceivedDateStr = (poData[i][7] instanceof Date) ? Utilities.formatDate(poData[i][7], timeZone, "yyyy-MM-dd") : String(poData[i][7]);
        }
        let unitCost = 0;
        if (poData[i].length > 9 && poData[i][9] !== "") {
          unitCost = parseFloat(poData[i][9]) || 0;
        }
        let totalCost = 0;
        if (poData[i].length > 10 && poData[i][10] !== "") {
          totalCost = parseFloat(poData[i][10]) || 0;
        }
        let rowSupplier = "";
        if (poData[i].length > 11 && poData[i][11] !== "") {
          rowSupplier = String(poData[i][11]).trim();
        }

        purchaseOrders.push({
          poNumber: poData[i][0],
          prNumber: poData[i][1],
          productId: poData[i][2],
          productName: poData[i][3],
          orderDate: orderDateStr,
          orderedQty: parseFloat(poData[i][5] || 0),
          receivedQty: parseFloat(poData[i][6] || 0),
          lastReceivedDate: lastReceivedDateStr,
          status: poData[i][8] || '',
          unitCost: unitCost,
          totalCost: totalCost,
          supplier: rowSupplier
        });
      }
    }
  }

  const appData = {
    products: products,
    machines: machines,
    mappings: mappings,
    settings: settings,
    manuals: manuals,
    lots: getLotsFromJson(),
    purchaseOrders: purchaseOrders
  };

  try {
    const folder = DriveApp.getFolderById(JSON_FOLDER_ID);
    const files = folder.getFilesByName("app_data.json");
    const content = JSON.stringify(appData);
    if (files.hasNext()) {
      const file = files.next();
      file.setContent(content);
    } else {
      folder.createFile("app_data.json", content, MimeType.PLAIN_TEXT);
    }
  } catch (e) {
    Logger.log("rebuildAndSaveAppDataJson error: " + e.toString());
  }

  // ซิงค์ข้อมูลไปยัง Firebase
  try {
    syncToFirebase(appData);
  } catch (e) {
    Logger.log("Firebase Sync error: " + e.toString());
    if (throwError) {
      throw new Error("Firebase Sync failed: " + e.message);
    }
  }

  return appData;
}

// ฟังก์ชันซิงค์ข้อมูลไปยัง Firebase Realtime Database
function syncToFirebase(appData) {
  if (!FIREBASE_URL || FIREBASE_URL === "") {
    Logger.log("Firebase URL is not configured.");
    return;
  }
  
  let baseUrl = FIREBASE_URL;
  if (baseUrl.charAt(baseUrl.length - 1) !== '/') {
    baseUrl += '/';
  }
  
  const url = baseUrl + "appData.json" + (FIREBASE_SECRET ? "?auth=" + FIREBASE_SECRET : "");
  
  const options = {
    method: "put",
    contentType: "application/json",
    payload: JSON.stringify(appData),
    muteHttpExceptions: true
  };
  
  const response = UrlFetchApp.fetch(url, options);
  const responseCode = response.getResponseCode();
  if (responseCode !== 200) {
    throw new Error("Failed to sync to Firebase: HTTP " + responseCode + " - " + response.getContentText());
  }
}

  let baseUrl = FIREBASE_URL;

    baseUrl += '/';
  }
  const url = baseUrl + path + ".json" + (FIREBASE_SECRET ? "?auth=" + FIREBASE_SECRET : "");
  const response = UrlFetchApp.fetch(url, { method: "get", muteHttpExceptions: true });
  if (response.getResponseCode() === 200) {
    return JSON.parse(response.getContentText());
  }
  return null;
}

// ฟังก์ชันบันทึกข้อมูลไปยัง Firebase Realtime Database
function setFirebaseData(path, data) {
  if (!FIREBASE_URL || FIREBASE_URL === "") {
    throw new Error("FIREBASE_URL is not configured.");
  }
  let baseUrl = FIREBASE_URL;
  if (baseUrl.charAt(baseUrl.length - 1) !== '/') {
    baseUrl += '/';
  }
  const url = baseUrl + path + ".json" + (FIREBASE_SECRET ? "?auth=" + FIREBASE_SECRET : "");
  const options = {
    method: "put",
    contentType: "application/json",
    payload: JSON.stringify(data),
    muteHttpExceptions: true
  };
  const response = UrlFetchApp.fetch(url, options);
  if (response.getResponseCode() !== 200) {
    throw new Error("Failed to write to Firebase: " + response.getContentText());
  }
}

// ฟังก์ชันอัปโหลดภาพและสร้างลิงก์ lh3 (รองรับการเก็บในโฟลเดอร์ย่อยเพื่อความเป็นระเบียบ)
function uploadImageAndGetLh3Url(base64Data, fileName, subfolderName) {
  try {
    if (!base64Data) return "";
    const base64 = base64Data.split(',')[1] || base64Data;
    const decoded = Utilities.base64Decode(base64);
    const blob = Utilities.newBlob(decoded, "image/jpeg", fileName); 
    
    let folder = DriveApp.getFolderById(DRIVE_FOLDER_ID);
    if (subfolderName) {
      const subfolders = folder.getFoldersByName(subfolderName);
      if (subfolders.hasNext()) {
        folder = subfolders.next();
        // ลบไฟล์เก่าออกก่อนอัปโหลดใหม่
        const oldFiles = folder.getFiles();
        while (oldFiles.hasNext()) {
          oldFiles.next().setTrashed(true);
        }
      } else {
        folder = folder.createFolder(subfolderName);
      }
    }
    
    const file = folder.createFile(blob);
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    
    return 'https://lh3.googleusercontent.com/d/' + file.getId();
  } catch (e) {
    throw new Error("อัปโหลดรูปภาพล้มเหลว: " + e.message);
  }
}

// 7. เพิ่ม Mapping
function addMapping(data) {
  let productIds = Array.isArray(data.product_ids) ? data.product_ids : [data.product_id];
  let machineId = String(data.machine_id);
  
  let mappings = getMappingsFromJson();
  let isModified = false;
  for (let p = 0; p < productIds.length; p++) {
    let pid = String(productIds[p]);
    if(!pid) continue;
    let isDuplicate = mappings.some(m => String(m.product_id) === pid && String(m.machine_id) === machineId);
    if (!isDuplicate) {
      mappings.push({ product_id: pid, machine_id: machineId });
      isModified = true;
    }
  }
  
  if (isModified) {
    saveMappingsToJson(mappings);
  } else {
    throw new Error("รายการอะไหล่ที่เลือก ถูกจับคู่กับเครื่องจักรนี้อยู่แล้วทั้งหมด");
  }
}

  }

function migrateTransactionsToJson() {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const txSheet = ss.getSheetByName("Transactions");
  const txdSheet = ss.getSheetByName("Transaction_Details");
  
  if (!txSheet || !txdSheet) return [];
  
  const txValues = txSheet.getDataRange().getValues();
  const txdValues = txdSheet.getDataRange().getValues();
  const detailsMap = {};
  
  if (txdValues.length > 1) {
    for (let i = 1; i < txdValues.length; i++) {
      const r = txdValues[i];
      const txId = String(r[1]).trim();
      if (!detailsMap[txId]) detailsMap[txId] = [];
      detailsMap[txId].push({
        detail_id: r[0], product_id: r[2], qty: r[3], price: r[4], subtotal: r[5]
      });
    }
  }
  
  const transactions = [];
  if (txValues.length > 1) {
    for (let i = 1; i < txValues.length; i++) {
      const r = txValues[i];
      let dateVal = r[1];
      if (dateVal instanceof Date) {
        dateVal = Utilities.formatDate(dateVal, Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss");
      }
      transactions.push({
        id: String(r[0]).trim(),
        date: dateVal,
        requester: r[2],
        department: r[3],
        machine_id: r[4],
        total_price: r[5],
        note: r[6],
        status: r[7],
        items: detailsMap[String(r[0]).trim()] || []
      });
    }
  }
  
  saveTransactionsToJson(transactions);
  return transactions;
}

// 9. บันทึกการทำรายการเบิกอะไหล่ (POS Checkout) พร้อมตัดสต็อกตาม FIFO
function checkoutOrder(payload) {
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(30000);
  } catch (e) {
    throw new Error("ระบบไม่ว่างชั่วคราว กรุณาลองใหม่อีกครั้ง");
  }
  
  try {
    // ดึงข้อมูลทั้งหมดในครั้งเดียว (ประหยัด HTTP Request)
    const db = getFirebaseData("") || {};
    let products = db.appData?.products || [];
    const cart = payload.cart; 
    let lots = db.lots || [];
    let transactions = db.transactions || [];
    
    const prodMap = {};
    products.forEach(p => {
      prodMap[p.id] = p;
    });
    
    // ตรวจเช็คสต็อกคงเหลือ
    for (let k = 0; k < cart.length; k++) {
      const item = cart[k];
      const prodId = String(item.id).trim();
      const product = prodMap[prodId];
      if (!product) throw new Error("ไม่พบอะไหล่รหัส " + prodId + " ในระบบ");
      
      const stockQty = parseFloat(product.stock_qty) || 0;
      if (stockQty < item.qty) {
        throw new Error("สต็อกไม่เพียงพอ: อะไหล่ " + product.name + " (" + prodId + ") มีคงเหลือ " + stockQty + " ชิ้น แต่พยายามเบิก " + item.qty + " ชิ้น");
      }
      
      const prodLots = lots.filter(l => String(l.product_id).trim() === prodId && l.remaining_qty > 0);
      const totalLotQty = prodLots.reduce((sum, l) => sum + l.remaining_qty, 0);
      if (totalLotQty < stockQty) {
        const diff = stockQty - totalLotQty;
        lots.push({
          lot_id: "LOT-SUPP-" + prodId + "-" + Date.now(),
          product_id: prodId,
          cost: parseFloat(product.cost) || 0,
          price_a: parseFloat(product.price_a) || 0,
          price_b: parseFloat(product.price_b) || 0,
          price_c: parseFloat(product.price_c) || 0,
          initial_qty: diff,
          remaining_qty: diff,
          created_at: Utilities.formatDate(new Date(), "Asia/Bangkok", "yyyy-MM-dd HH:mm:ss"),
          note: "Lot สำรองคงเหลือ"
        });
      }
    }
    
    // ตัดสต็อกตาม FIFO (First-In, First-Out)
    const checkoutItems = [];
    for (let k = 0; k < cart.length; k++) {
      const item = cart[k];
      const prodId = String(item.id).trim();
      let neededQty = item.qty;
      
      const availableLots = lots.filter(l => String(l.product_id).trim() === prodId && l.remaining_qty > 0)
                                .sort((a, b) => new Date(a.created_at || 0) - new Date(b.created_at || 0));
      
      for (let l = 0; l < availableLots.length && neededQty > 0; l++) {
        const currentLot = availableLots[l];
        const takeQty = Math.min(neededQty, currentLot.remaining_qty);
        
        currentLot.remaining_qty -= takeQty;
        neededQty -= takeQty;
        
        let lotPrice = item.price;
        if (item.priceLevel === 'A' && currentLot.price_a) lotPrice = currentLot.price_a;
        else if (item.priceLevel === 'B' && currentLot.price_b) lotPrice = currentLot.price_b;
        else if (item.priceLevel === 'C' && currentLot.price_c) lotPrice = currentLot.price_c;
        
        checkoutItems.push({
          detail_id: "",
          product_id: prodId,
          lot_id: currentLot.lot_id,
          qty: takeQty,
          unit_cost: currentLot.cost || 0,
          price: lotPrice,
          subtotal: takeQty * lotPrice
        });
      }
      
      const product = prodMap[prodId];
      product.stock_qty = (parseFloat(product.stock_qty) || 0) - item.qty;
    }
    
    const now = new Date();
    const dateStr = Utilities.formatDate(now, "Asia/Bangkok", "yyyy-MM-dd HH:mm:ss");
    const datePrefix = Utilities.formatDate(now, "Asia/Bangkok", "yyyyMMdd");
    
    let counter = 1;
    if (transactions.length > 0) {
      const lastTx = transactions[transactions.length - 1];
      if (String(lastTx.id).indexOf("LDT-" + datePrefix) === 0) {
        const parts = String(lastTx.id).split("-");
        const lastNum = parseInt(parts[2], 10);
        if (!isNaN(lastNum)) counter = lastNum + 1;
      }
    }
    const txId = "LDT-" + datePrefix + "-" + String(counter).padStart(4, '0');
    
    let calcTotalPrice = 0;
    checkoutItems.forEach((it, idx) => {
      it.detail_id = txId + "-" + (idx + 1);
      calcTotalPrice += it.subtotal;
    });
    
    const newTransaction = {
      id: txId,
      date: dateStr,
      requester: payload.requester,
      department: payload.department,
      machine_id: payload.machine_id,
      serial_number: payload.serial_number || "",
      total_price: payload.total_price || calcTotalPrice,
      note: payload.note || "",
      status: "Success",
      items: checkoutItems
    };
    
    transactions.push(newTransaction);
    
    // อัปเดตข้อมูล Firebase ทั้งหมดในครั้งเดียวด้วย PATCH (ลดการเรียกจาก 3 ครั้งเหลือ 1 ครั้ง)
    const updates = {
      "appData/products": products,
      "lots": lots,
      "transactions": transactions
    };
    patchFirebaseData("", updates);
    
    return { transaction_id: txId, items: checkoutItems };
  } finally {
    lock.releaseLock();
  }
}

// 10. ดึงข้อมูลประวัติใบเบิกอะไหล่ทั้งหมด
function getTransactions() {
  let transactions = getTransactionsFromJson();
  return transactions.reverse();
}

function cancelTransaction(payload) {
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(30000);
  } catch (e) {
    throw new Error("ระบบไม่ว่างชั่วคราว กรุณาลองใหม่อีกครั้ง");
  }
  
  try {
    const txId = String(payload.transaction_id).trim();
    let transactions = getTransactionsFromJson();
    
    const txIndex = transactions.findIndex(t => t.id === txId);
    if (txIndex === -1) throw new Error("ไม่พบรหัสใบเบิก " + txId);
    
    const targetTx = transactions[txIndex];
    if (targetTx.status === "Cancelled") throw new Error("ใบเบิกนี้ถูกยกเลิกไปแล้ว");
    
    let products = getFirebaseData("appData/products") || [];
    let lots = getLotsFromJson();
    
    for (let k = 0; k < targetTx.items.length; k++) {
      const item = targetTx.items[k];
      const prodId = String(item.product_id).trim();
      
      const product = products.find(p => String(p.id).trim() === prodId);
      if (product) {
        product.stock_qty = (parseFloat(product.stock_qty) || 0) + item.qty;
      }
      
      if (item.lot_id) {
        const targetLot = lots.find(l => l.lot_id === item.lot_id);
        if (targetLot) {
          targetLot.remaining_qty += item.qty;
        }
      }
    }

function hashPassword(password) {
  const digest = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, password, Utilities.Charset.UTF_8);
  let hexString = "";
  for (let i = 0; i < digest.length; i++) {
    let byteVal = digest[i];
    if (byteVal < 0) byteVal += 256;
    let byteString = byteVal.toString(16);
    if (byteString.length == 1) byteString = "0" + byteString;
    hexString += byteString;
  }
  return hexString;
}

function restockProduct(payload) {
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(30000);
  } catch (e) {
    throw new Error("ระบบไม่ว่างชั่วคราว กรุณาลองใหม่อีกครั้ง");
  }
  
  try {
    // ดึงข้อมูลทั้งหมดในครั้งเดียว
    const db = getFirebaseData("") || {};
    let products = db.appData?.products || [];
    let lots = db.lots || [];
    let transactions = db.transactions || [];
    
    const product = products.find(p => String(p.id).trim() === String(payload.id).trim());
    if (!product) throw new Error("ไม่พบอะไหล่รหัส " + payload.id + " ในระบบ");
    
    const qty = parseFloat(payload.qty) || 0;
    const cost = (payload.cost !== undefined && payload.cost !== "") ? parseFloat(payload.cost) : product.cost;
    const pA = (payload.price_a !== undefined && payload.price_a !== "") ? parseFloat(payload.price_a) : product.price_a;
    const pB = (payload.price_b !== undefined && payload.price_b !== "") ? parseFloat(payload.price_b) : product.price_b;

    const lotId = "LOT-" + payload.id + "-" + Date.now();
    lots.push({
      lot_id: lotId,
      product_id: payload.id,
      cost: cost,
      price_a: pA,
      price_b: pB,
      price_c: pC,
      initial_qty: qty,
      remaining_qty: qty,
      created_at: Utilities.formatDate(new Date(), "Asia/Bangkok", "yyyy-MM-dd HH:mm:ss"),
      note: payload.note || "เติมสต็อกสินค้า"
    });
    
    product.stock_qty = (parseFloat(product.stock_qty) || 0) + qty;
    product.cost = cost;
    product.price_a = pA;
    product.price_b = pB;
    product.price_c = pC;
    
    let lastTxId = 0;
    if (transactions.length > 0) {
      const lastTx = transactions[transactions.length - 1];
      lastTxId = parseInt(String(lastTx.id).replace("TX-", "")) || 0;
    }
    const nextTxId = "TX-" + String(lastTxId + 1).padStart(6, '0');
    
    const newRestockTransaction = {
      id: nextTxId,
      requester: payload.requester,
      department: payload.department || "สโตร์ (Restock)",
      machine_id: "RESTOCK",
      serial_number: "",
      total_price: 0,
      note: payload.note || "เติมสต็อกสินค้า",
      created_at: Utilities.formatDate(new Date(), "Asia/Bangkok", "yyyy-MM-dd HH:mm:ss"),
      status: "Restock",
      items: [{
        lot_id: lotId,
        product_id: payload.id,
        qty: qty,
        cost: cost,
        price: 0
      }]
    };
    
    transactions.push(newRestockTransaction);
    
    // อัปเดตข้อมูล Firebase ในครั้งเดียวด้วย PATCH
    const updates = {
      "appData/products": products,
      "lots": lots,
      "transactions": transactions
    };
    patchFirebaseData("", updates);
    
    return { new_stock: product.stock_qty, transaction_id: nextTxId, lot_id: lotId };

    lock.releaseLock();
  }

function editPurchaseOrderDraft(payload) {
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(30000);
  } catch (e) {
    throw new Error("ระบบไม่ว่างชั่วคราว กรุณาลองใหม่อีกครั้ง");
  }
  
  try {
    const poNumber = String(payload.poNumber || '').trim();
    const productId = String(payload.productId || '').trim();
    const orderedQty = parseFloat(payload.orderedQty) || 0;
    if (orderedQty <= 0) throw new Error("จำนวนที่สั่งต้องมากกว่า 0");
    
    let purchaseOrders = getFirebaseData("appData/purchaseOrders") || [];
    let index = -1;
    for (let i = 0; i < purchaseOrders.length; i++) {
      const o = purchaseOrders[i];
      if (poNumber && poNumber !== "") {
        if (String(o.poNumber).trim() === poNumber) {
          index = i;
          break;
        }
      } else {
        if (String(o.productId).trim() === productId && String(o.status).trim() === "เตรียมสั่ง") {
          index = i;
          break;
        }
      }
    }
    
    if (index === -1) throw new Error("ไม่พบรายการใบสั่งซื้อที่ต้องการแก้ไข");

    }
    
    purchaseOrders[index].orderedQty = orderedQty;
    purchaseOrders[index].totalCost = orderedQty * (parseFloat(purchaseOrders[index].unitCost) || 0);
    
    setFirebaseData("appData/purchaseOrders", purchaseOrders);
    
    return { poNumber: poNumber, productId: productId, orderedQty: orderedQty };
  } finally {
    lock.releaseLock();
  }

    for (let i = 0; i < purchaseOrders.length; i++) {
      const o = purchaseOrders[i];
      if (poNumber && poNumber !== "") {
        if (String(o.poNumber).trim() === poNumber) {
          index = i;
          break;
        }
      } else {
        if (String(o.productId).trim() === productId && String(o.status).trim() === "เตรียมสั่ง") {
          index = i;
          break;
        }
      }
    }
    
    if (index === -1) throw new Error("ไม่พบรายการใบสั่งซื้อที่ต้องการลบ");
    if (String(purchaseOrders[index].status).trim() !== "เตรียมสั่ง") {
      throw new Error("สามารถลบได้เฉพาะรายการที่มีสถานะเตรียมสั่งเท่านั้น");
    }
    
    purchaseOrders.splice(index, 1);
    setFirebaseData("appData/purchaseOrders", purchaseOrders);
    
    return { poNumber: poNumber, productId: productId, deleted: true };
  } finally {
    lock.releaseLock();
  }
}

// 12.5 อัปเดตข้อมูลใบสั่งซื้อในหน้าจัดการคำสั่งซื้อ
function updatePurchaseOrderDraft(payload) {
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(30000);
  } catch (e) {
    throw new Error("ระบบไม่ว่างชั่วคราว กรุณาลองใหม่อีกครั้ง");
  }
  
  try {
    const originalPoNumber = String(payload.originalPoNumber).trim();
    const newPoNumber = String(payload.newPoNumber || '').trim();
    const newPrNumber = String(payload.newPrNumber || '').trim();
    const orderedQty = parseFloat(payload.orderedQty) || 0;
    const unitCost = parseFloat(payload.unitCost) || 0;
    const status = String(payload.status || '').trim();
    const productId = String(payload.productId).trim();
    const newSupplier = String(payload.newSupplier || '').trim();
    const newUnit = String(payload.newUnit || '').trim();
    
    if (orderedQty <= 0) throw new Error("จำนวนที่สั่งต้องมากกว่า 0");
    
    let purchaseOrders = getFirebaseData("appData/purchaseOrders") || [];
    const index = purchaseOrders.findIndex(o => String(o.poNumber).trim() === originalPoNumber);
    if (index === -1) throw new Error("ไม่พบรายการใบสั่งซื้อต้นฉบับ " + originalPoNumber);
    
    let currentStatus = String(purchaseOrders[index].status).trim();
    if (status) {
      currentStatus = status;
    } else if (newPoNumber && newPoNumber.indexOf("PO-DRF-") !== 0) {
      currentStatus = "สั่งแล้ว";
    }
    
    purchaseOrders[index] = {
      poNumber: newPoNumber || originalPoNumber,
      prNumber: newPrNumber || "PR-DRAFT",
      productId: productId,
      productName: purchaseOrders[index].productName,
      orderDate: purchaseOrders[index].orderDate,
      orderedQty: orderedQty,
      receivedQty: parseFloat(purchaseOrders[index].receivedQty) || 0,
      lastReceivedDate: purchaseOrders[index].lastReceivedDate || "",
      status: currentStatus,
      unitCost: unitCost,
      totalCost: orderedQty * unitCost,
      supplier: newSupplier
    };
    
    // Update master product record if found
    let products = getFirebaseData("appData/products") || [];
    const prodIndex = products.findIndex(p => String(p.id).trim() === productId);
    if (prodIndex !== -1) {
      products[prodIndex].cost = unitCost;
      products[prodIndex].supplier = newSupplier;
      if (newUnit) {
        products[prodIndex].unit = newUnit;
      }

      // Auto-pricing update based on new cost
      let factorA = 1.05;
      let factorB = 1.10;
      let factorC = 1.20;
      if (unitCost >= 10000) { factorA = 1.02; factorB = 1.05; factorC = 1.10; }
      else if (unitCost >= 5000) { factorA = 1.03; factorB = 1.07; factorC = 1.15; }
      
      products[prodIndex].price_a = Math.ceil(unitCost * factorA);
      products[prodIndex].price_b = Math.ceil(unitCost * factorB);
      products[prodIndex].price_c = Math.ceil(unitCost * factorC);
      
      setFirebaseData("appData/products", products);
    }
    
    setFirebaseData("appData/purchaseOrders", purchaseOrders);
    
    if (newSupplier) {
      let products = getFirebaseData("appData/products") || [];
      const prodIndex = products.findIndex(p => String(p.id).trim() === productId);
      if (prodIndex !== -1) {
        products[prodIndex].supplier = newSupplier;
        setFirebaseData("appData/products", products);
      }
    }
    
    return {
      poNumber: newPoNumber || originalPoNumber,
      status: currentStatus
    };
  } finally {
    lock.releaseLock();
  }
}

// ฟังก์ชันอัปโหลดไฟล์คู่มือ (PDF หรือ รูปภาพ) ลงใน Google Drive
function uploadManualFileToDrive(base64Data, fileName, mimeType) {
  try {
    if (!base64Data) return "";
    // ถ้าเป็น URL อยู่แล้ว (เช่น ลิงก์เดิม) ไม่ต้องอัพโหลดซ้ำ
    if (base64Data.indexOf('http://') === 0 || base64Data.indexOf('https://') === 0) {
      return base64Data;
    }
    const parts = base64Data.split(',');
    let type = mimeType || "application/pdf";
    if (parts.length > 1 && parts[0].indexOf(';base64') !== -1) {
      const match = parts[0].match(/data:(.*?);base64/);
      if (match && match[1]) type = match[1];
    }
    const base64 = parts[1] || base64Data;
    const decoded = Utilities.base64Decode(base64);
    
    // ตั้งนามสกุลไฟล์ตามประเภท
    let ext = ".pdf";
    if (type.indexOf('image/png') !== -1) ext = ".png";
    else if (type.indexOf('image/jpeg') !== -1 || type.indexOf('image/jpg') !== -1) ext = ".jpg";
    else if (type.indexOf('image/webp') !== -1) ext = ".webp";
    
    const finalFileName = (fileName || ("manual_" + Date.now())) + ext;
    const blob = Utilities.newBlob(decoded, type, finalFileName);
    
    const folder = DriveApp.getFolderById(DRIVE_FOLDER_ID);
    const file = folder.createFile(blob);
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    
    // ลิงก์ดาวน์โหลดตรงจาก Google Drive
    return 'https://drive.google.com/uc?export=download&id=' + file.getId();
  } catch (e) {
    Logger.log("อัปโหลดไฟล์คู่มือลง Google Drive ล้มเหลว: " + e.toString());
    return base64Data;
  }
}

function addManual(payload) {
  const nowStr = Utilities.formatDate(new Date(), "Asia/Bangkok", "yyyy-MM-dd");
  const id = payload.id || ("MAN-" + String(Date.now()).slice(-6));
  
  let fileUrl = payload.file_url || "";
  if (fileUrl && fileUrl.indexOf("data:") === 0) {
    const uploadedUrl = uploadManualFileToDrive(fileUrl, (payload.title || "manual") + "_" + id, payload.file_type);
    if (uploadedUrl) fileUrl = uploadedUrl;
  }
  
  let manuals = getFirebaseData("appData/manuals") || [];
  manuals.push({
    id: id,
    title: payload.title || "",
    description: payload.description || "",
    file_url: fileUrl,
    file_type: payload.file_type || "",
    uploaded_at: nowStr
  });
  setFirebaseData("appData/manuals", manuals);
  return { id: id, file_url: fileUrl };
}

function editManual(payload) {
  let manuals = getFirebaseData("appData/manuals") || [];
  const index = manuals.findIndex(m => String(m.id).trim() === String(payload.id).trim());
  if (index === -1) throw new Error("ไม่พบคู่มือที่ต้องการแก้ไข");
  
  let fileUrl = payload.file_url || "";
  if (fileUrl && fileUrl.indexOf("data:") === 0) {
    const uploadedUrl = uploadManualFileToDrive(fileUrl, (payload.title || "manual") + "_" + payload.id, payload.file_type);
    if (uploadedUrl) fileUrl = uploadedUrl;
  } else {
    fileUrl = manuals[index].file_url;
  }
  
  manuals[index] = {
    id: payload.id,
    title: payload.title || "",
    description: payload.description || "",
    file_url: fileUrl,
    file_type: payload.file_type || manuals[index].file_type,
    uploaded_at: manuals[index].uploaded_at
  };
  setFirebaseData("appData/manuals", manuals);
  return { status: "success", file_url: fileUrl };
}

      for (let i = 1; i < prodData.length; i++) {
        let r = prodData[i];
        products.push({
          id: r[0], name: r[1], unit: r[2], cost: parseFloat(r[3]) || 0,
          price_a: parseFloat(r[4]) || 0, price_b: parseFloat(r[5]) || 0, price_c: parseFloat(r[6]) || 0,
          category: r[7], note: r[8], image_url: r[9],
          stock_qty: (r[10] !== undefined && r[10] !== "") ? parseFloat(r[10]) : 0,
          group: r[11] || "", supplier: r[12] || "", storage: r[13] || ""
        });
      }
    }
    setFirebaseData("appData/products", products);
    
    // 2. อ่าน Machines
    const machSheet = ss.getSheetByName("Machines");
    const machData = machSheet.getDataRange().getValues();
    const machines = [];
    if (machData.length > 1) {
      for (let i = 1; i < machData.length; i++) {
        let r = machData[i];
        machines.push({ 
          id: r[0], name: r[1], image_url: r[2] || '', cost: parseFloat(r[3]) || 0,
          price_a: parseFloat(r[4]) || 0, price_b: parseFloat(r[5]) || 0, price_c: parseFloat(r[6]) || 0,
          note: r[7] || '', group: r[8] || '', supplier: r[9] || '', storage: r[10] || ''
        });
      }
    }
    setFirebaseData("appData/machines", machines);
    
    // 3. อ่าน Settings
    const settingsSheet = ss.getSheetByName("Settings");
    const settings = { isShowPriceBForGuest: false, isShowPriceCForGuest: false };
    if (settingsSheet) {
      const settingsData = settingsSheet.getDataRange().getValues();
      if (settingsData.length > 1) {
        for (let i = 1; i < settingsData.length; i++) {
          let key = settingsData[i][0];
          let val = settingsData[i][1];
          if (key === "isShowPriceBForGuest") settings.isShowPriceBForGuest = (String(val).toLowerCase() === "true");
          if (key === "isShowPriceCForGuest") settings.isShowPriceCForGuest = (String(val).toLowerCase() === "true");
        }
      }
    }
    setFirebaseData("appData/settings", settings);
    
    // 4. อ่าน Manuals
    const manSheet = ss.getSheetByName("Manuals");
    const manuals = [];
    if (manSheet) {
      const manData = manSheet.getDataRange().getValues();
      if (manData.length > 1) {
        for (let i = 1; i < manData.length; i++) {
          manuals.push({
            id: manData[i][0], title: manData[i][1], description: manData[i][2] || '',
            file_url: manData[i][3] || '', file_type: manData[i][4] || '', uploaded_at: manData[i][5] || ''
          });
        }
      }
    }
    setFirebaseData("appData/manuals", manuals);
    
    // 5. อ่าน PurchaseOrders
    const poSheet = ss.getSheetByName("PurchaseOrders");
    const purchaseOrders = [];
    if (poSheet) {
      const poData = poSheet.getDataRange().getValues();
      if (poData.length > 1) {
        const timeZone = ss.getSpreadsheetTimeZone();
        for (let i = 1; i < poData.length; i++) {
          let orderDateStr = poData[i][4] instanceof Date ? Utilities.formatDate(poData[i][4], timeZone, "yyyy-MM-dd") : String(poData[i][4]);
          let lastReceivedDateStr = poData[i][7] instanceof Date ? Utilities.formatDate(poData[i][7], timeZone, "yyyy-MM-dd") : String(poData[i][7]);
          purchaseOrders.push({
            poNumber: poData[i][0], prNumber: poData[i][1], productId: poData[i][2], productName: poData[i][3],
            orderDate: orderDateStr, orderedQty: parseFloat(poData[i][5] || 0), receivedQty: parseFloat(poData[i][6] || 0),
            lastReceivedDate: lastReceivedDateStr, status: poData[i][8] || '',
            unitCost: parseFloat(poData[i][9]) || 0, totalCost: parseFloat(poData[i][10]) || 0, supplier: poData[i][11] || ""
          });
        }
      }
    }
    setFirebaseData("appData/purchaseOrders", purchaseOrders);
    
    // 6. อ่าน Transactions จาก Sheet
    const txSheet = ss.getSheetByName("Transactions");
    if (txSheet) {
      const txData = txSheet.getDataRange().getValues();
      const transactions = [];
      if (txData.length > 1) {
        for (let i = 1; i < txData.length; i++) {
          let items = [];
          try {
            items = JSON.parse(txData[i][9] || "[]");
          } catch(e) {}
          transactions.push({
            id: txData[i][0], requester: txData[i][1], department: txData[i][2], machine_id: txData[i][3],
            serial_number: txData[i][4], total_price: parseFloat(txData[i][5]) || 0, note: txData[i][6],
            created_at: String(txData[i][7]), status: txData[i][8], items: items
          });
        }
        setFirebaseData("transactions", transactions);
      }
    }

    // 7. อ่าน Mappings จาก Sheet
    const mapSheet = ss.getSheetByName("Mappings");
    if (mapSheet) {
      const mapData = mapSheet.getDataRange().getValues();
      const mappings = [];
      if (mapData.length > 1) {
        for (let i = 1; i < mapData.length; i++) {
          mappings.push({
            product_id: String(mapData[i][0]).trim(),
            machine_id: String(mapData[i][1]).trim()
          });
        }
        setFirebaseData("mappings", mappings);
      }
    }

    // 8. อ่าน Lots จาก Sheet
    const lotsSheet = ss.getSheetByName("Lots");
    if (lotsSheet) {
      const lotsData = lotsSheet.getDataRange().getValues();
      const lots = [];
      if (lotsData.length > 1) {
        for (let i = 1; i < lotsData.length; i++) {
          lots.push({
            lot_id: String(lotsData[i][0]).trim(),
            product_id: String(lotsData[i][1]).trim(),
            cost: parseFloat(lotsData[i][2]) || 0,
            price_a: parseFloat(lotsData[i][3]) || 0,
            price_b: parseFloat(lotsData[i][4]) || 0,
            price_c: parseFloat(lotsData[i][5]) || 0,
            initial_qty: parseFloat(lotsData[i][6]) || 0,
            remaining_qty: parseFloat(lotsData[i][7]) || 0,
            created_at: String(lotsData[i][8]),
            note: String(lotsData[i][9])
          });
        }
        setFirebaseData("lots", lots);
      }
    }

    // 9. อ่าน Users จาก Sheet
    const usersSheet = ss.getSheetByName("Users");
    if (usersSheet) {
      const usersData = usersSheet.getDataRange().getValues();
      const users = [];
      if (usersData.length > 1) {
        for (let i = 1; i < usersData.length; i++) {
          users.push({
            fullName: String(usersData[i][0]).trim(),
            department: String(usersData[i][1]).trim(),
            phone: String(usersData[i][2]).trim(),
            email: String(usersData[i][3]).trim().toLowerCase(),
            passwordHash: String(usersData[i][4]).trim(),
            role: String(usersData[i][5]).trim() || "user",
            priceLevel: String(usersData[i][6]).trim() || "A",
            userType: String(usersData[i][7]).trim() || "insource"
          });
        }
        setFirebaseData("users", users);
      }
    }

    ui.alert('✅ สำเร็จ', 'อัปโหลดข้อมูลจาก Google Sheet ขึ้น Firebase เรียบร้อยแล้ว', ui.ButtonSet.OK);
  } catch (e) {
    ui.alert('❌ ผิดพลาด', 'เกิดข้อผิดพลาดในการอัปโหลด: ' + e.toString(), ui.ButtonSet.OK);
  }
}
